/* ============ بيانات مشتركة ============ */
async function fetchBoardsWithStatus(){
  const {data: boards, error} = await sb.from('billboards').select('*').order('created_at', {ascending: false}).limit(1000);
  if (error) throw error;
  let status = {};
  if (isAdmin()) {
    const t = today();
    const {data: cs, error: ce} = await sb.from('contracts').select('id,billboard_id,company_name,start_date,end_date').gte('end_date', t).order('start_date').limit(2000);
    if (ce) throw ce;
    for (const c of cs) {
      const s = status[c.billboard_id];
      if (c.start_date <= t) status[c.billboard_id] = {k: 'rented', c};
      else if (!s) status[c.billboard_id] = {k: 'reserved', c};
    }
  }
  return {boards, status};
}
function statusTag(st){
  if (!isAdmin()) return '';
  if (!st) return `<span class="tag green">متاحة</span>`;
  if (st.k === 'rented') return `<span class="tag orange">مؤجّرة</span>`;
  return `<span class="tag blue">محجوزة من ${dateAr(st.c.start_date)}</span>`;
}
function boardItem(b, st){
  return `<div class="item" onclick="go('board',{id:${b.id}})">
    <div class="ic">🪧</div>
    <div class="body">
      <div class="t">${esc(b.governorate)} — ${esc(b.address)}</div>
      <div class="s">📐 ${fmt(b.width)} × ${fmt(b.height)} م &nbsp;·&nbsp; 🏢 ${esc(b.owner_name)}</div>
      ${isAdmin() ? `<div style="margin-top:6px">${statusTag(st)}${st && st.k === 'rented' ? ` <span class="s">${esc(st.c.company_name)} حتى ${dateAr(st.c.end_date)}</span>` : ''}</div>` : ''}
    </div><span class="chev">‹</span></div>`;
}
async function fetchOpenPayments(){
  const {data, error} = await sb.from('payments')
    .select('id,seq,amount,due_date,paid,contract_id,contracts(id,company_name,campaign_name,currency,payments_count,customer_phone,billboards(governorate,address))')
    .eq('paid', false).order('due_date').limit(1000);
  if (error) throw error;
  return data;
}
function payItem(p){
  const c = p.contracts || {}, t = today();
  const late = p.due_date < t, soon = !late && p.due_date <= addDays(t, 2);
  const tag = late ? `<span class="tag red">متأخرة ${diffDays(p.due_date, t)} يوم</span>`
    : p.due_date === t ? `<span class="tag orange">اليوم</span>`
    : soon ? `<span class="tag orange">خلال 48 ساعة</span>` : `<span class="tag gray">${dateAr(p.due_date)}</span>`;
  return `<div class="item" onclick="go('contract',{id:${p.contract_id}})">
    <div class="ic" style="background:${late ? 'var(--red-bg)' : 'var(--green-bg)'}">💵</div>
    <div class="body"><div class="t">${esc(c.company_name)} — ${esc(c.campaign_name)}</div>
      <div class="s">الدفعة ${p.seq} من ${c.payments_count} · <b style="color:var(--text)">${money(p.amount, c.currency)}</b> · ${dateAr(p.due_date)}</div>
      <div style="margin-top:6px">${tag}</div></div><span class="chev">‹</span></div>`;
}

function onDates(){
  const s = $('#c_s').value, e = $('#c_e').value;
  $('#c_d').value = durationText(s, e);
  const f = document.querySelector('form[onsubmit^="saveContract"]');
  const m = f && f.getAttribute('onsubmit').match(/saveContract\(event, (\d+), (\d+)\)/);
  if (m) checkOverlap(+m[1], +m[2]);
}
async function checkOverlap(bid, cid){
  const s = $('#c_s')?.value, e = $('#c_e')?.value; const w = $('#c_warn'); if (!s || !e || !w) return;
  let q = sb.from('contracts').select('company_name,start_date,end_date').eq('billboard_id', bid).lte('start_date', e).gte('end_date', s);
  if (cid) q = q.neq('id', cid);
  const {data} = await q;
  w.innerHTML = data && data.length ? `<div class="msg err">⚠️ اللوحة محجوزة بنفس الفترة: ${data.map(x => `${esc(x.company_name)} (${dateAr(x.start_date)} ← ${dateAr(x.end_date)})`).join('، ')}</div>` : '';
}
function rebuildSched(){
  const v = Math.max(0, Math.round(Number($('#c_v').value) || 0));
  const n = Math.min(120, Math.max(1, parseInt($('#c_n').value) || 1));
  const first = $('#c_f').value || $('#c_s').value || today();
  const every = Math.max(1, parseInt($('#c_every')?.value) || 1);
  const base = Math.floor(v / n);
  SCHED = Array.from({length: n}, (_, i) => ({amount: base + (i === n - 1 ? v - base * n : 0), due_date: addMonths(first, i * every)}));
  drawSched();
}
function drawSched(){
  const el = $('#sched'); if (!el) return;
  const cur = $('#c_cur').value;
  el.innerHTML = SCHED.map((p, i) => `<div class="sched-row"><div class="n">${i + 1}</div>
    <input type="number" min="0" step="1" inputmode="numeric" value="${p.amount}" oninput="SCHED[${i}].amount=Number(this.value)||0;sumSched()">
    <input type="date" value="${p.due_date}" onchange="SCHED[${i}].due_date=this.value"></div>`).join('') + `<div id="ssum"></div>`;
  sumSched();
}
function sumSched(){
  const v = Math.round(Number($('#c_v').value) || 0), cur = $('#c_cur').value;
  const t = SCHED.reduce((a, p) => a + Number(p.amount), 0);
  const ok = t === v;
  $('#ssum').innerHTML = `<div class="sumline" style="background:${ok ? 'var(--green-bg)' : 'var(--red-bg)'};color:${ok ? 'var(--green)' : 'var(--red)'}">
    <span>مجموع الدفعات</span><span>${money(t, cur)} ${ok ? '✓' : `(الفرق ${money(v - t, cur)})`}</span></div>`;
}
async function saveContract(e, bid, id){
  e.preventDefault();
  const btn = $('#c_btn'), m = $('#c_msg'); m.classList.add('hidden');
  const c = {
    company_name: $('#c_co').value.trim(), campaign_name: $('#c_ca').value.trim(),
    start_date: $('#c_s').value, end_date: $('#c_e').value,
    duration_text: $('#c_d').value.trim() || durationText($('#c_s').value, $('#c_e').value),
    customer_phone: $('#c_ph').value.trim(), contract_value: Math.round(Number($('#c_v').value) || 0),
    currency: $('#c_cur').value
  };
  const fail = t => { m.textContent = t; m.classList.remove('hidden'); };
  if (c.end_date < c.start_date) return fail('تاريخ النهاية قبل تاريخ البداية');
  if (phoneIntl(c.customer_phone).length < 11) return fail('رقم الهاتف غير صحيح');
  if (!id) {
    if (SCHED.reduce((a, p) => a + Number(p.amount), 0) !== c.contract_value) return fail('مجموع الدفعات لازم يساوي قيمة العقد');
    if (SCHED.some(p => !p.due_date)) return fail('كل دفعة تحتاج تاريخ');
  }
  btn.disabled = true; btn.textContent = 'جاري الحفظ...';
  try {
    if (id) {
      const {error} = await sb.from('contracts').update(c).eq('id', id);
      if (error) throw error;
      toast('✅ تم تعديل العقد'); goBack();
    } else {
      const {data: newId, error} = await sb.rpc('create_contract_with_payments', {c: {...c, billboard_id: bid, payments_count: SCHED.length}, p: SCHED});
      if (error) throw error;
      toast('✅ تم حفظ العقد');
      CURRENT = {name: 'contract', params: {id: newId}}; render();
    }
  } catch (err) { fail(errMsg(err)); btn.disabled = false; btn.textContent = '💾 حفظ العقد'; }
}

      </div>
    </div>
    <div class="card"><h3>💵 الدفعات</h3>
      ${c.payments.map(p => {
        const late = !p.paid && p.due_date < t;
        return `<div class="pay ${p.paid ? 'paid' : late ? 'late' : ''}">
          <div class="num">${p.paid ? '✓' : p.seq}</div>
          <div class="b"><div class="a">${money(p.amount, c.currency)}</div>
            <div class="d">${p.paid ? `دُفعت ${dateAr(new Date(p.paid_at).toLocaleDateString('en-CA', {timeZone: 'Asia/Baghdad'}))}` : `الموعد ${dateAr(p.due_date)}${late ? ` · <b style="color:var(--red)">متأخرة ${diffDays(p.due_date, t)} يوم</b>` : ''}`}</div></div>

function waText(c, p, kind){
  const cur = c.currency;
  if (kind === 'hello') {
    const next = c.payments.find(x => !x.paid);
    return `مرحباً ${c.company_name} 🌟\nبخصوص حملة "${c.campaign_name}"` + (next ? `\nالدفعة القادمة: ${money(next.amount, cur)} بتاريخ ${dateAr(next.due_date)}` : '');
  }
  if (kind === 'remind') return `مرحباً ${c.company_name} 🌟\nنذكّركم بموعد الدفعة رقم (${p.seq}) لحملة "${c.campaign_name}"\nالمبلغ: ${money(p.amount, cur)}\nالموعد: ${dateAr(p.due_date)}\nشكراً لكم 🙏`;
  const next = c.payments.filter(x => !x.paid && x.id !== p.id).sort((a, b) => a.seq - b.seq)[0];
  let t = `مرحباً ${c.company_name} 🌟\nتم دفع الدفعة (${money(p.amount, cur)})\nحملة "${c.campaign_name}" — الدفعة ${p.seq} من ${c.payments_count}`;
  t += next ? `\n\nالدفعة القادمة: ${money(next.amount, cur)}\nموعدها: ${dateAr(next.due_date)}` : `\n\n✅ تم تسديد كامل مبلغ العقد`;
  return t + `\n\nشكراً لتعاملكم معنا 🙏`;
}
function openWA(cid, pid, kind){
  const c = window._contract; if (!c || c.id !== cid) return;
  const p = pid ? c.payments.find(x => x.id === pid) : null;
  const text = waText(c, p, kind);
  const url = `https://wa.me/${phoneIntl(c.customer_phone)}?text=${encodeURIComponent(text)}`;
  modal(`<h3>${kind === 'paid' ? '✅ تم تسجيل الدفع' : '💬 رسالة للزبون'}</h3>
    <div class="hint">راح تنفتح الرسالة جاهزة على رقم الزبون ${esc(c.customer_phone)}، بس اضغط إرسال.</div>
    <div class="wa-preview">${esc(text)}</div>
    <a class="btn wa" href="${url}" target="_blank" rel="noopener" onclick="closeModal()">📲 إرسال عبر واتساب</a>
    <button class="btn light" style="margin-top:10px" onclick="closeModal()">لاحقاً</button>`);
}
async function markPaid(pid){
  const c = window._contract, p = c.payments.find(x => x.id === pid);
  if (!confirm(`تأكيد استلام الدفعة ${p.seq}: ${money(p.amount, c.currency)}؟`)) return;
  const {error} = await sb.from('payments').update({paid: true, paid_at: new Date().toISOString(), paid_by: ME.id}).eq('id', pid);
  if (error) return toast('⚠️ ' + errMsg(error));
  await sb.from('notifications').update({is_read: true}).eq('payment_id', pid);
  p.paid = true;
  await SCREENS.contract({id: c.id});
  openWA(c.id, pid, 'paid');
  refreshBell();
}

SCREENS.payments = async ({tab = 'soon'}) => {
  shell('الدفعات', '', loading(), {tab: 'payments'});
  const open = await fetchOpenPayments();
  const t = today();
  const groups = {
    late: open.filter(p => p.due_date < t),
    soon: open.filter(p => p.due_date >= t && p.due_date <= addDays(t, 7)),
    all: open.filter(p => p.due_date >= t)
  };
  window._payGroups = groups;
  $('#main').innerHTML = `<div class="tabs" id="ptabs">
      <button data-k="late" onclick="showPay('late')">متأخرة (${groups.late.length})</button>
      <button data-k="soon" onclick="showPay('soon')">خلال 7 أيام (${groups.soon.length})</button>
      <button data-k="all" onclick="showPay('all')">كل القادمة (${groups.all.length})</button>
      <button data-k="paid" onclick="showPay('paid')">المستلَمة</button></div><div id="plist"></div>`;
  showPay(tab);
};
async function showPay(k){
  document.querySelectorAll('#ptabs button').forEach(b => b.classList.toggle('on', b.dataset.k === k));
  CURRENT.params.tab = k;
  const el = $('#plist');
  if (k === 'paid') {
    el.innerHTML = loading();
    const {data, error} = await sb.from('payments').select('id,seq,amount,due_date,paid_at,contract_id,contracts(company_name,campaign_name,currency,payments_count)').eq('paid', true).order('paid_at', {ascending: false}).limit(100);
    if (error) { el.innerHTML = `<div class="msg err">${esc(errMsg(error))}</div>`; return; }
    el.innerHTML = data.length ? data.map(p => `<div class="item" onclick="go('contract',{id:${p.contract_id}})"><div class="ic" style="background:var(--green-bg)">✅</div>
      <div class="body"><div class="t">${esc(p.contracts.company_name)} — ${esc(p.contracts.campaign_name)}</div>
      <div class="s">الدفعة ${p.seq} من ${p.contracts.payments_count} · <b style="color:var(--green)">${money(p.amount, p.contracts.currency)}</b> · استُلمت ${dateAr(new Date(p.paid_at).toLocaleDateString('en-CA', {timeZone: 'Asia/Baghdad'}))}</div></div><span class="chev">‹</span></div>`).join('')
      : `<div class="empty"><div class="e">💵</div>لا توجد دفعات مستلمة بعد</div>`;
    return;
  }
  const list = window._payGroups[k] || [];
  el.innerHTML = list.length ? list.map(payItem).join('') : `<div class="empty"><div class="e">✅</div>لا توجد دفعات هنا</div>`;
}

async function callManage(body){
  const {data, error} = await sb.functions.invoke('manage-user', {body});
  if (error) {
    let m = error.message;
    try { const j = await error.context.json(); if (j.error) m = j.error; } catch (_) {}
    throw new Error(m);
  }
  if (data && data.error) throw new Error(data.error);
  return data;
}
async function saveUser(e, id){
  e.preventDefault();
  const btn = $('#u_btn'), m = $('#u_msg'); m.classList.add('hidden');
  const body = {full_name: $('#u_n').value.trim(), email: $('#u_e').value.trim().toLowerCase(), password: $('#u_p').value, role: $('#u_r').value};
  if (body.password && body.password.length < 6) { m.textContent = 'كلمة المرور 6 أحرف على الأقل'; m.classList.remove('hidden'); return; }
  btn.disabled = true; btn.textContent = 'جاري الحفظ...';
  try {
    if (id) { if (!body.password) delete body.password; if (id === ME.id) delete body.role; await callManage({action: 'update', id, ...body}); }
    else await callManage({action: 'create', ...body});
    if (id === ME.id) ME.full_name = body.full_name;
    toast('✅ تم حفظ الحساب'); goBack();
  } catch (err) { m.textContent = errMsg(err); m.classList.remove('hidden'); btn.disabled = false; btn.textContent = '💾 حفظ الحساب'; }
}
async function deleteUser(id){
  if (!confirm('حذف هذا الحساب نهائياً؟ (اللوحات اللي أضافها تبقى)')) return;
  try { await callManage({action: 'delete', id}); toast('تم حذف الحساب'); goBack(); }
  catch (err) { toast('⚠️ ' + errMsg(err)); }
}
